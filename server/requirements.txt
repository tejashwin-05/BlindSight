import os
import sys
import subprocess

def _start_backend():
    env = os.environ.copy()
    env["ECOSIGHT_HEADLESS"] = "1"
    env["ECOSIGHT_SERVER_ONLY"] = "1"
    script_dir = os.path.dirname(os.path.abspath(__file__))
    main_py = os.path.join(script_dir, "main.py")
    backend = subprocess.Popen(
        [sys.executable, main_py],
        cwd=script_dir,
        env=env,
    )
